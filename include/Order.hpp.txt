#pragma once
#include <cstdint>

enum class Side : uint8_t {
    BUY,
    SELL
};

enum class OrderType : uint8_t {
    LIMIT,
    MARKET
};

struct Order {
    uint64_t orderId;
    uint64_t timestamp;
    double price;
    uint32_t quantity;
    Side side;
    OrderType type;

    Order(uint64_t id, Side s, double p, uint32_t q, uint64_t ts = 0, OrderType t = OrderType::LIMIT)
        : orderId(id), timestamp(ts), price(p), quantity(q), side(s), type(t) {}
};

struct Trade {
    uint64_t buyOrderId;
    uint64_t sellOrderId;
    double price;
    uint32_t quantity;
    uint64_t timestamp;
};
